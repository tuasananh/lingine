\chapter*{Tài liệu tham khảo}
\addcontentsline{toc}{chapter}{Tài liệu tham khảo}

\begin{enumerate}
    \item \textbf{Đại học Bách khoa Hà Nội}, \textit{Bài giảng Trí tuệ Nhân tạo IT3160}, PGS.TS Lê Thanh Hương, Trường Công nghệ Thông tin và Truyền thông.
    \item \textbf{Russell, S. và Norvig, P.}, \textit{Artificial Intelligence: A Modern Approach}, Prentice Hall, 4th Edition, 2020 (Tài liệu về mô hình tác tử PEAS và thuật toán Alpha-Beta).
    \item \textbf{Chess Programming Wiki (CPW)}, \url{https://www.chessprogramming.org/} (Tài liệu tham khảo chi tiết về Bitboards, Magic Bitboards, Transposition Table, PVS, NMP, LMR, và Singular Extensions).
    \item \textbf{Steve J. Edwards}, \textit{An Introduction to Zobrist Hashing}, 1996 (Cơ chế sinh khóa và cập nhật vi phân mã băm).
    \item \textbf{Peter Bradley và John Terry}, \textit{Rank Analysis of Incomplete Block Designs: I. The Method of Paired Comparisons}, Biometrika, 1952 (Mô hình Bradley-Terry dùng ước lượng sức mạnh Elo).
    \item \textbf{Peter Österlund}, \textit{Texel Tuning Method}, \url{https://github.com/peterosterlund/texel} (Phương pháp tối ưu hóa tham số đánh giá thế cờ bằng hồi quy logistic).
    \item \textbf{The Rust Programming Language}, Steve Klabnik và Carol Nichols, No Starch Press, 2018 (Tài liệu hướng dẫn tối ưu hiệu năng và an toàn bộ nhớ trong Rust).
    \item \textbf{Fairy-Stockfish Project}, \url{https://github.com/ianfab/Fairy-Stockfish} (Công cụ đối thủ chuẩn dùng cho cờ tướng và các biến thể cờ vua).
\end{enumerate}
