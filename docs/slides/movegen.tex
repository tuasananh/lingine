\begin{frame}{Nhóm bị cản sử dụng Magic Bitboards}
  Ở đây ta lấy ví dụ với quân Mã, quân Tượng làm tương tự.
  \begin{itemize}[<+->]
    \item Mã có tối đa 8 nước.
    \item Nước nhảy phụ thuộc 4 ô chân Mã.
    \item $\Rightarrow$ Có $2^4=16$ cách chặn khác nhau tại mỗi vị trí trên bàn cờ.
    \item $\Rightarrow$ Sinh ra toàn bộ các trường hợp có thể xảy ra.
  \end{itemize}
  \onslide<5->{\begin{alertblock}{Vấn đề}
    Mỗi một trường hợp chặn được biểu diễn trên Bitboard có thể có giá trị rất lớn.

    \onslide<6->{$\Rightarrow$ Cần có một cách ánh xạ chúng về 1 chỉ số từ $0$ đến $15$.}
\end{alertblock}}
\end{frame}
