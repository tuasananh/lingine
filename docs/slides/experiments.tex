% ================= FRAME 2 =================
\begin{frame}{Hiệu năng và Tốc độ sinh nước đi}
    \begin{itemize}
        \item<1-> \textbf{Kiến trúc tối ưu:} Sử dụng bàn cờ \textit{Bitboard} và kỹ thuật tra cứu bảng tĩnh (\textit{Magic Bitboards}).
        \item<2-> \textbf{Đo lường thực tế:}
        \begin{itemize}
            \item \textbf{Khối lượng:} Duyệt hơn \textbf{5.3 tỷ} trạng thái (độ sâu $d=6$).
            \item \textbf{Thời gian:} Hoàn thành trong vỏn vẹn \textbf{54.65 giây} (Intel Core i9-13900HX).
        \end{itemize}
        \item<3-> \textbf{Tốc độ trung bình:}
        \begin{center}
            \Large \textbf{\textcolor{blue}{$\approx 98.68$ triệu nút / giây}}
        \end{center}
        \item<4-> \textbf{Kết luận:} Loại bỏ hoàn toàn vòng lặp quét tia truyền thống, thay bằng phép toán thao tác bit song song trên thanh ghi 64-bit $\rightarrow$ Đạt hiệu năng cực hạn.
    \end{itemize}
\end{frame}

% ================= FRAME 4 =================
\begin{frame}{3 Giai đoạn Kiểm định \& Cấu hình đấu}
    \textbf{Kiểm định nghiêm ngặt qua 3 giai đoạn độc lập:}
    \vspace{0.2cm}
    \begin{enumerate}
        \item<1-> \textbf{Gauntlet Performance:} Đấu 3000 ván với 6 cấp độ \textit{Fairy-Stockfish} (ELO từ 1200 đến 2200). Đổi màu quân luân phiên để triệt tiêu lợi thế đi trước.
        \item<2-> \textbf{Neighbor Match:} Đấu 1000 ván với phiên bản tiền nhiệm ($N$ vs $N-1$) đo lường mức cải tiến ELO trực tiếp.
        \item<3-> \textbf{Base Match:} Đấu 1000 ván với bản sơ khai ($1.0.0$) đo tổng tiến bộ lũy kế.
    \end{enumerate}
    
    \vspace{0.4cm}
    \onslide<4->{
        \textbf{Cấu hình môi trường:}
        \begin{itemize}
            \item \textit{Kiểm soát thời gian:} 3 giây + 0.03 giây/nước.
            \item \textit{Đa dạng hóa:} Dùng \textit{Masters Opening Database} (40,711 ván) tạo thế cờ ngẫu nhiên, tránh lặp ván.
        \end{itemize}
    }
\end{frame}
