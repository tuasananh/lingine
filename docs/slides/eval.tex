\begin{frame}{Vì sao cần heuristic?}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.50\textwidth}
      Search không thể duyệt tới kết thúc ván trong mọi lượt.
      \begin{itemize}
        \item Cây trò chơi bùng nổ theo depth.
        \item Node lá cần được lượng hóa bằng điểm số.
        \item Điểm dương nghiêng về Đỏ, điểm âm nghiêng về Đen.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Heuristic}
        \[
          \eval(S)\approx \text{lợi thế của Đỏ so với Đen}
        \]
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Tapered evaluation}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Trung cuộc và tàn cuộc cần trọng số khác nhau.
        \item Lingine pha trộn theo \code{phase}.
        \item Xe, Pháo, Mã ảnh hưởng phase nhiều hơn Tốt/Tướng.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Công thức}
        \[
          Score=\frac{MG\cdot phase+EG\cdot(32-phase)}{32}
        \]
      \end{block}
      \centering
      \begin{fitdiagram}
