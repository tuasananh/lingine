% FRAME 2: TOÁN HỌC & TRIỂN KHAI
\begin{frame}{Mô hình Toán học \& Tối ưu hóa}
    Mục tiêu: Tối thiểu hóa sai số toàn phương trung bình (\textbf{MSE}) để tìm bộ tham số $\vec{\theta}$:
    \vspace{-0.2cm}
    \begin{equation*}
        \text{MSE}(\vec{\theta}) = \frac{1}{N} \sum_{i=1}^{N} \left( R_i - \sigma\left(E(S_i, \vec{\theta})\right) \right)^2 \quad \text{với} \quad \sigma(x) = \frac{1}{1 + 10^{-\frac{K \cdot x}{400}}}
    \end{equation*}

    Trong đó:
\begin{itemize}
    \item<2-> $R_i \in \{1.0, 0.5, 0.0\}$ là kết quả thực tế của ván đấu chứa thế cờ thứ $i$.
    \item<3-> $E(S_i, \vec{\theta})$ là hàm đánh giá tĩnh thế cờ $S_i$ bằng bộ tham số $\vec{\theta}$.
    \item<4-> $\sigma(x)$ là hàm kích hoạt Sigmoid có nhiệm vụ ánh xạ điểm số cờ sang $[0, 1]$ đại diện cho cơ hội chiến thắng dự kiến.
\end{itemize}

\onslide<5->{\begin{block}{Giải pháp}
Sử dụng các thuật toán tối ưu để tìm lời giải cho bài toán trên, Lingine đã sử dụng Adam Optimizer.
\end{block}}
\vspace{0.5cm}

\end{frame}
