                    \item Thế trận rủi ro hơn dự kiến.
                    \item Mở rộng đáy: $\alpha_{new} = \alpha - \delta$.
                    \item Duyệt lại: \textbf{Giữ nguyên độ sâu}.
                \end{itemize}
                
                \vspace{0.4cm}
                % Kịch bản Fail-high
                \item<2-> \textcolor{red}{\textbf{3. Fail-high}} ($\text{score} \ge \beta$):
                \begin{itemize}
                    \item Nước đi áp đảo (\textit{Fail-high Reductions}).
                    \item Mở rộng đỉnh: $\beta_{new} = \beta + \delta$.

% ================= FRAME 1: CÁC MỞ RỘNG CƠ BẢN =================
\begin{frame}{Các kỹ thuật Mở rộng không gian (Search Extensions)}
    \textbf{Mục tiêu:} Bù đắp độ sâu tại các thời điểm nhạy cảm để tránh \textit{Hiệu ứng chân trời} (Horizon Effect).
    
    \begin{itemize}
        \item<2-> \textbf{1. Mở rộng khi bị chiếu (Check Extensions)}
        \begin{itemize}
            \item \textit{Bối cảnh:} Mọi nước đi đều bắt buộc, dễ dẫn đến biến động vật chất lớn.
            \item \textit{Xử lý:} Nếu Tướng đang bị chiếu $\Rightarrow$ Tăng độ sâu ($d \leftarrow d + 1$) để phòng thủ hoặc tìm sát cục.
        \end{itemize}
        
        \item<3-> \textbf{2. Mở rộng nước đi duy nhất (One-Reply Extensions)}
        \begin{itemize}
            \item \textit{Bối cảnh:} Nút mạng chỉ có đúng 1 nước đi hợp lệ.
            \item \textit{Đặc điểm:} Đào sâu thêm 1 ply gần như \textbf{miễn phí} tài nguyên (hệ số rẽ nhánh = 1).
            \item \textit{Xử lý:} Bổ sung tầm nhìn chiến thuật bằng cách tăng độ sâu ($d \leftarrow d + 1$).
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Cắt tỉa nước đi muộn (Late Move Pruning - LMP)}
    \begin{itemize}
        \item<1-> \textbf{Ý tưởng:} Ở độ sâu thấp, nếu thử một lượng lớn các nước đi bình thường (\textit{quiet moves}) mà không cải thiện được $\alpha$ $\rightarrow$ Có thể kết luận các nước xếp cuối bảng (chưa được duyệt) đều vô giá trị.

\begin{frame}{Giảm độ sâu nước đi muộn (Late Move Reductions - LMR)}
    \begin{itemize}
        \item<1-> \textbf{Ý tưởng:} Các nước đi trễ (kém hấp dẫn) không bị ngắt hoàn toàn bởi LMP sẽ tiếp tục được duyệt nhưng với \textbf{độ sâu giảm bớt}.
