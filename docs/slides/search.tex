\begin{frame}{Negamax Alpha-Beta}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.42\textwidth}
      \[
        V(s)=\max_{a\in\actions(s)} -V(Result(s,a))

      \begin{itemize}
        \item Negamax gộp MAX/MIN bằng đổi dấu.
        \item Alpha-Beta cắt nhánh không ảnh hưởng quyết định cuối.
        \item Fail-soft trả về bound chính xác hơn.
      \end{itemize}

\begin{frame}{PVS và aspiration window}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Principal Variation Search}
        Nước đầu tiên search full window; các nước sau dùng null window để xác nhận chúng không vượt PV.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Aspiration Window}
        Dùng score depth trước làm trung tâm cửa sổ hẹp; fail high/low thì mở rộng.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \centering
  \begin{fitdiagram}

\end{frame}

\begin{frame}{Quiescence search}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Xử lý horizon effect.
        \item Nếu đang bị chiếu: sinh mọi nước né chiếu.
        \item Nếu không bị chiếu: chỉ sinh capture.
        \item Dừng ở thế tương đối tĩnh.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{fitdiagram}
        \begin{tikzpicture}[node distance=0.5cm]
          \node[flowbox] (leaf) {Depth = 0};
          \node[flowbox, below=of leaf] (capture) {Còn capture/chiếu?};
          \node[flowbox, below left=0.6cm and 0.45cm of capture] (more) {Search tiếp};
          \node[redbox, below right=0.6cm and 0.45cm of capture] (evaln) {Static Eval};

    \end{column}
  \end{columns}
\end{frame}
