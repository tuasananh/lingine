\begin{frame}{Trạng thái trung cuộc}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.44\textwidth}
      Đặc điểm của trạng thái trung cuộc:
      \begin{itemize}
        \item Số lượng quân còn lại ảnh hưởng trực tiếp đến hệ số nhánh (branching factor).
        \item Lịch sử nước đi cần thiết để kiểm tra trạng thái lặp và thực hiện hoàn tác (undo).
        \item Hàm lượng giá (evaluation) tích hợp cả điểm vật chất lẫn vị trí quân cờ.
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[height=0.65\textheight]{\figpath{figure-01c-master-midgame.png}}
      \figcaption{2.c}{Một state trung cuộc.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Bitboard \code{u128}}
  \begin{columns}[T,totalwidth=\textwidth]

\begin{frame}{Zobrist hashing}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \[
        H(S)=Z_{side}\oplus\bigoplus_{sq=0}^{89} Z_{piece(sq),sq}
      \]
      \begin{itemize}
        \item Mỗi cặp quân--ô có khóa 64-bit.
        \item Cập nhật hash vi phân sau mỗi move.
        \item Hash dùng cho repetition và Transposition Table.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{fitdiagram}
        \begin{tikzpicture}[node distance=0.45cm]
          \node[flowbox] (old) {$H_{old}$};
          \node[flowbox, below=of old] (a) {$\oplus K_{piece,from}$};
          \node[flowbox, below=of a] (b) {$\oplus K_{piece,to}$};
          \node[flowbox, below=of b] (c) {$\oplus K_{side}$};
          \node[redbox, below=of c] (new) {$H_{new}$};
          \draw[arrow] (old) -- (a);
          \draw[arrow] (a) -- (b);
          \draw[arrow] (b) -- (c);
          \draw[arrow] (c) -- (new);
        \end{tikzpicture}
      \end{fitdiagram}
    \end{column}
  \end{columns}
\end{frame}
